# Run these commands in your PowerShell window:

# Step 2: Install dependencies (takes 3-5 minutes)
npm install

# Step 3: Install Firebase CLI globally
npm install -g firebase-tools

# Step 4: Login to Firebase (opens browser)
firebase login

# Step 5: Build web version (takes 5-10 minutes)
npx expo export:web

# Step 6: Deploy to Firebase
firebase deploy --only hosting

# Your app will be live at: https://parrot-1fc71.web.app
